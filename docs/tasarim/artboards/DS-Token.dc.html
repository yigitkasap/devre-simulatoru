/* Gerilim → renk (önerilen varsayılan) */
function wireColor(v, vmax) {
  const x = Math.max(-1, Math.min(1, v / vmax));
  const hue = x >= 0 ? 140 - 140 * x : 140 + 80 * -x;
  const light = hue > 45 && hue <= 200 ? 32 : 40;
  return `hsl(${hue} 75% ${light}%)`;
}
